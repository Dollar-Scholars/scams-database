# nginx site for the Scams DB. Copy to /etc/nginx/sites-available/ and enable it
# (see docs/DEPLOY.md). Certbot adds the HTTPS part to this file.
server {
    listen 80;
    listen [::]:80;
    server_name scamdb.dollarscholars.org;

    client_max_body_size 1m;

    location / {
        proxy_pass http://127.0.0.1:8001;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        # Must always overwrite (never pass on) the client's value: Django trusts it
        # when DJANGO_SECURE_PROXY_SSL_HEADER=True.
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_read_timeout 30s;
    }
}
